cmake_minimum_required(VERSION 3.24)
project(regime_retrieval LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-fno-fast-math -ffp-contract=off)
find_package(OpenMP REQUIRED)
find_library(LZ4_LIBRARY lz4 REQUIRED)
set(FAISS_SOURCE_DIR "" CACHE PATH "Clean, pinned FAISS checkout (the build copies and patches it)")
if(NOT EXISTS "${FAISS_SOURCE_DIR}/faiss/impl/HNSW.cpp")
  message(FATAL_ERROR "Set FAISS_SOURCE_DIR to the pinned FAISS checkout in README.md")
endif()
file(SHA256 "${FAISS_SOURCE_DIR}/faiss/impl/HNSW.cpp" upstream_hnsw)
if(NOT upstream_hnsw STREQUAL "55378e9f68e3ae6d585773ba126f31b82e4892f5f09420d1f1d389b9a3082992")
  message(FATAL_ERROR "The FAISS HNSW source differs from the pinned revision")
endif()
set(faiss_copy "${CMAKE_BINARY_DIR}/faiss-src")
file(MAKE_DIRECTORY "${faiss_copy}")
file(COPY "${FAISS_SOURCE_DIR}/" DESTINATION "${faiss_copy}" PATTERN ".git" EXCLUDE PATTERN "build" EXCLUDE)
configure_file("${FAISS_SOURCE_DIR}/faiss/impl/HNSW.cpp" "${faiss_copy}/faiss/impl/HNSW.cpp" COPYONLY)
configure_file(index/native_hop.hpp "${faiss_copy}/faiss/impl/native_hop.hpp" COPYONLY)
find_program(PATCH_EXECUTABLE patch REQUIRED)
execute_process(COMMAND "${PATCH_EXECUTABLE}" --batch --forward -p0 -i "${CMAKE_CURRENT_SOURCE_DIR}/index/native_hop.patch"
                WORKING_DIRECTORY "${faiss_copy}/faiss/impl" COMMAND_ERROR_IS_FATAL ANY)
foreach(opt FAISS_ENABLE_GPU FAISS_ENABLE_METAL FAISS_ENABLE_PYTHON FAISS_ENABLE_C_API FAISS_ENABLE_EXTRAS FAISS_ENABLE_MKL BUILD_TESTING)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory("${faiss_copy}" faiss-build EXCLUDE_FROM_ALL)
foreach(target inspect_index build_index build_shards search shard_search)
  add_executable(${target} index/${target}.cpp)
  target_link_libraries(${target} PRIVATE faiss OpenMP::OpenMP_CXX)
endforeach()
find_package(JNI REQUIRED)
add_library(regime_time SHARED iotdb/time_jni.cpp)
target_include_directories(regime_time PRIVATE storage ${JNI_INCLUDE_DIRS})
target_link_libraries(regime_time PRIVATE "${LZ4_LIBRARY}")
target_compile_definitions(regime_time PRIVATE TS_RESTART)
add_library(pack SHARED storage/pack.cpp)
add_executable(read_records storage/read_records.cpp)
target_link_libraries(read_records PRIVATE "${LZ4_LIBRARY}")
target_compile_definitions(read_records PRIVATE TS_RESTART)
add_executable(public_first32 experiments/public_datasets/public_first32.cpp)
add_executable(components experiments/components.cpp)
target_link_libraries(components PRIVATE "${LZ4_LIBRARY}")
target_compile_definitions(components PRIVATE TS_RESTART)

add_executable(ob_router oceanbase/router.cpp)
target_include_directories(ob_router PRIVATE index)
target_link_libraries(ob_router PRIVATE faiss OpenMP::OpenMP_CXX)
add_executable(ob_pages oceanbase/pages.cpp)
target_include_directories(ob_pages PRIVATE storage)
target_link_libraries(ob_pages PRIVATE "${LZ4_LIBRARY}")
target_compile_definitions(ob_pages PRIVATE TS_RESTART)
